{{- /* Social share image (1200x630) for a post: the base card in assets/og/ with the post title drawn on it at build time. Returns the image resource, or nil for non-post pages (they use the site default). */ -}} {{- $img := "" -}} {{- if and .IsPage (eq .Section "posts") -}} {{- $base := resources.Get "og/base.png" -}} {{- $font := resources.Get "og/DejaVuSans-Bold.ttf" -}} {{- $title := .Title -}} {{- $size := cond (gt (len $title) 70) 50 60 -}} {{- /* images.Text wraps at the image's right edge, so draw the title on a 1040px-wide transparent canvas and overlay it at x=80: 80px margins. */ -}} {{- $canvas := resources.Get "og/text-canvas.png" -}} {{- $text := $canvas | images.Filter (images.Text $title (dict "color" "#ffffff" "size" $size "linespacing" 16 "x" 0 "y" 0 "font" $font)) -}} {{- $img = $base | images.Filter (images.Overlay $text 80 150) -}} {{- end -}} {{- return $img -}}
